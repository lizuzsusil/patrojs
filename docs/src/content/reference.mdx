### Core Types

#### `PatroJsDateProps`

```ts
interface PatroJsDateProps {
  year: number;  // BS year (2000–2099)
  month: number; // 1–12
  day: number;   // 1–32
}
```

#### `PatroJsDay`

```ts
interface PatroJsDay {
  day: number;
  isToday: boolean;
  isSelected: boolean;
  isDisabled: boolean;
  adDate: Date;  // Equivalent AD date
}
```

#### `PatroJsMonth`

```ts
interface PatroJsMonth {
  year: number;
  month: number;
  monthName: string;
  weekDays: string[];
  days: (PatroJsDay | null)[];
  allMonths: string[];
}
```

#### `PatroJsLanguage`

```ts
type PatroJsLanguage = 'en' | 'ne';
```

#### `PatroJsFormatter`

```ts
interface PatroJsFormatter {
  format(date: PatroJsDateProps, language: PatroJsLanguage): string;
  parse(value: string, language: PatroJsLanguage): PatroJsDateProps | null;
  isValid(value: string, language: PatroJsLanguage): boolean;
  placeholder(language: PatroJsLanguage): string;
}
```

```tsx
<PatroDatePicker formatPattern="YYYY/MM/DD" />
```

### Theme Configuration

The `config` prop customizes appearance without writing custom CSS.

```ts
interface PatroJsPickerConfig {
  size?: 'sm' | 'md' | 'lg';
  sizeTokens?: Partial<PatroJsSizeTokens>;
  colors?: ColorConfig;
  radius?: RadiusConfig;
  gridStyle?: 'borderless' | 'bordered';
  navIcons?: NavIconConfig;
}
```

#### Color Configuration

```ts
interface ColorConfig {
  primary?: string;      // Selected day bg, today border, footer btn bg
  primaryBg?: string;    // Text color on primary background
  todayBorder?: string;  // Override today's border color
}
```

#### Radius Configuration

```ts
interface RadiusConfig {
  calendar?: string;  // Dropdown panel
  input?: string;     // Input field
  day?: string;       // Day cells
  select?: string;    // Month/year selects
  navBtn?: string;    // Nav buttons
  todayBtn?: string;  // Today button
}
```

#### Example - green theme, compact size, pill input

```tsx
<PatroDatePicker
  config={{
    size: 'sm',
    colors: { primary: '#16a34a' },
    radius: { input: '999px', calendar: '16px', day: '8px' },
    gridStyle: 'bordered',
    navIcons: { prev: '◀', next: '▶' },
  }}
/>
```

## Style Slots

Override individual parts of the component using the `styles` or `classNames` props.

- `styles` accepts inline React `CSSProperties` objects.
- `classNames` accepts CSS class names for each component slot.
- Both props support the same slot keys.

<Tabs items={['Inline Styles', 'CSS Classes']}>
  <Tabs.Tab>

```tsx
import { PatroDatePicker } from '@patrojs/react';

export function StyledExample() {
  return (
    <PatroDatePicker
      styles={{
        input: { borderColor: '#2563eb', fontWeight: 600 },
        selectedDay: { backgroundColor: '#e11d48' },
      }}
    />
  );
}
```
