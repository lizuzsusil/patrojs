### Custom Formatter

<InteractiveExample code={`import type { PatroJsFormatter } from '@patrojs/react';

const customFormatter: PatroJsFormatter = {
  format: (date, lang) =>
    \`\${date.day}/\${date.month}/\${date.year}\`,
  parse: (value, lang) => {
    const parts = value.split('/');
    if (parts.length !== 3) return null;
    const [d, m, y] = parts.map(Number);
    return isNaN(y) || isNaN(m) || isNaN(d) ? null : { year: y, month: m, day: d };
  },
  isValid: (value, lang) => {
    const parsed = customFormatter.parse(value, lang);
    return parsed !== null && parsed.year >= 2000 && parsed.year <= 2099;
  },
  placeholder: (lang) => 'DD/MM/YYYY',
};

function CustomFormatterExample() {
  return <PatroDatePicker formatter={customFormatter} />;
}`}
tsxCode={`import { PatroDatePicker } from '@patrojs/react';
import type { PatroJsFormatter, PatroJsLanguage, PatroJsDateProps } from '@patrojs/core';

const customFormatter: PatroJsFormatter = {
  format: (date: PatroJsDateProps, lang: PatroJsLanguage): string =>
    \`\${date.day}/\${date.month}/\${date.year}\`,
  parse: (value: string, lang: PatroJsLanguage): PatroJsDateProps | null => {
    const parts = value.split('/');
    if (parts.length !== 3) return null;
    const [d, m, y] = parts.map(Number);
    return isNaN(y) || isNaN(m) || isNaN(d) ? null : { year: y, month: m, day: d };
  },
  isValid: (value: string, lang: PatroJsLanguage): boolean => {
    const parsed = customFormatter.parse(value, lang);
    return parsed !== null && parsed.year >= 2000 && parsed.year <= 2099;
  },
  placeholder: (lang: PatroJsLanguage): string => 'DD/MM/YYYY',
};

function CustomFormatterExample() {
  return <PatroDatePicker formatter={customFormatter} />;
}`}>
  <CustomFormatter />
</InteractiveExample>

### Next.js (App Router)

Wrap the import in a `'use client'` boundary:

```tsx
// app/components/PatroPicker.tsx
'use client';
export { PatroDatePicker } from '@patrojs/react';
```

Then use it in your server page:

```tsx
// app/page.tsx
import { PatroPicker } from './components/PatroPicker';
import '@patrojs/react/dist/index.css';

export default function Page() {
  return <PatroPicker />;
}
```
